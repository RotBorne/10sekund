-- 1. TABELA RODZINA
CREATE TABLE rodzina (
    id_rodziny SERIAL PRIMARY KEY,
    typ_rodziny VARCHAR(50),
    czas_stworzenia TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. TABELA UŻYTKOWNICY
CREATE TABLE uzytkownicy (
    id_uzytkownika SERIAL PRIMARY KEY,
    login VARCHAR(50) NOT NULL,
    typ_mieszkanca VARCHAR(50),
    czy_na_wyjezdzie BOOLEAN DEFAULT false,
    id_rodziny INT REFERENCES rodzina(id_rodziny) ON DELETE SET NULL
);

-- 3. TABELA LOGOWANIE
CREATE TABLE logowanie (
    login VARCHAR(50) PRIMARY KEY,
    haslo VARCHAR(255) NOT NULL,
    kod_rodziny VARCHAR(50)
);

-- 4. TABELA OD LISTY OBOWIĄZKÓW
CREATE TABLE lista_obowiazkow (
    id_zadania SERIAL PRIMARY KEY,
    nazwa_zadania VARCHAR(255) NOT NULL,
    czas_powtarzania VARCHAR(50),
    id_uzytkownika INT REFERENCES uzytkownicy(id_uzytkownika) ON DELETE CASCADE,
    czy_wykonane BOOLEAN DEFAULT false
);

-- 5. TABELA CZAS WYKONANIA
CREATE TABLE czas_wykonania (
    czas TIMESTAMP DEFAULT CURRENT_TIMESTAMP PRIMARY KEY,
    id_zadania INT REFERENCES lista_obowiazkow(id_zadania) ON DELETE CASCADE,
    id_uzytkownika INT REFERENCES uzytkownicy(id_uzytkownika) ON DELETE CASCADE
);
